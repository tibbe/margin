# Agent activity is announced in the window, and posted as one system
# notification per thread change while the document isn't in front;
# coming back clears them.
reset One two three.\n
save
select One
key cmd-opt-m
compose Why?
key cmd-enter
wait 100
select two
key cmd-opt-m
compose And this?
key cmd-enter
wait 100
# In front: announced only.
sh margin reply doc.md 1 'Because.'
wait-banner 1 new reply
banner
# In the background: each change is its own notification.
background on
sh margin reply doc.md 1 'Done.' --resolve
wait-banner comment resolved
banner
sh margin add doc.md --quote three 'Plural?'
wait-banner new comment
sh margin reopen doc.md 1
wait-banner reopened
sh margin delete doc.md 2
wait-banner deleted
banner
background off
comments
