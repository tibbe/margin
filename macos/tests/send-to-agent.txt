# Send to Agent reaches only an agent waiting with `margin wait`, and the
# toolbar follows it: waiting, then working until it waits again.
reset Retry three times.\n
save
agent
select three
key cmd-opt-m
compose Enough?
key cmd-enter
wait 100
# No agent waiting: nothing to send to.
agent
menu Comments > Send to Agent
key cmd-shift-enter
banner
sh margin wait doc.md > wait1.out 2>&1 &
wait 700
agent
menu Comments > Send to Agent
key cmd-shift-enter
wait-banner Sent
banner
wait 700
sh sed -E 's|`/[^`]*/|`|g' wait1.out
agent
sh margin reply doc.md 1 'Five, with backoff.'
wait-banner new reply
agent
# Waiting again ends working; a waiter that dies stops counting.
sh margin wait doc.md > wait2.out 2>&1 & echo $! > wait2.pid
wait 700
agent
sh kill $(cat wait2.pid)
wait 300
agent
